# Apply these response headers to every HTTPS response of the static app.
# This is a server-neutral reference; nginx.example.conf is the runnable sample.
# Default for HTML, SPA routes and /data/app-dataset.vault.json:

Cross-Origin-Opener-Policy: same-origin
Cross-Origin-Resource-Policy: same-origin
Permissions-Policy: camera=(), geolocation=(), microphone=(), payment=(), usb=()
Referrer-Policy: no-referrer
Strict-Transport-Security: max-age=31536000
X-Content-Type-Options: nosniff
X-Frame-Options: DENY
X-Permitted-Cross-Domain-Policies: none
X-Robots-Tag: noindex, nofollow, noarchive
